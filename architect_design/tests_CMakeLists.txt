include(FetchContent)

FetchContent_Declare(
    googletest
    GIT_REPOSITORY https://github.com/google/googletest.git
    GIT_TAG        v1.14.0
    GIT_SHALLOW    TRUE
)

# Prevent GTest from overriding our compiler/linker settings on Windows
set(gtest_force_shared_crt ON CACHE BOOL "" FORCE)
FetchContent_MakeAvailable(googletest)

enable_testing()

# ── Helper: one function wires up a test executable ──────────────────────────
function(add_core_test name source)
    add_executable(${name} ${source})
    target_link_libraries(${name} PRIVATE core GTest::gtest_main)
    apply_project_warnings(${name})
    add_test(NAME ${name} COMMAND ${name})

    # Each test binary runs from the project root so relative data/ paths work.
    set_tests_properties(${name} PROPERTIES
        WORKING_DIRECTORY ${PROJECT_SOURCE_DIR}
    )
endfunction()

# ── Foundation unit tests ─────────────────────────────────────────────────────
add_core_test(test_datetime     core/test_datetime.cpp)
add_core_test(test_idgenerator  core/test_idgenerator.cpp)
add_core_test(test_logger       core/test_logger.cpp)
add_core_test(test_config       core/test_config.cpp)
add_core_test(test_filehandler  core/test_filehandler.cpp)
add_core_test(test_integration  core/test_integration.cpp)
